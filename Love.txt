        function switchTab(tabName) {
            currentTab = tabName;

        async function initCamera() {
            const videoEl = document.getElementById('camera-stream');
            const errEl = document.getElementById('camera-error');
            try {
                if (mediaStream) mediaStream.getTracks().forEach(t => t.stop());
                mediaStream = await navigator.mediaDevices.getUserMedia({
                    video: { facingMode: useFrontCamera ? 'user' : 'environment', width: { ideal: 1080 }, height: { ideal: 1920 } },
                    audio: false
                });
                videoEl.srcObject = mediaStream;
                videoEl.classList.remove('hidden');
                errEl.classList.add('hidden');
            } catch (e) {
                console.warn("Camera fallback active:", e);
                videoEl.classList.add('hidden');
                errEl.classList.remove('hidden');
            }
        }

        function switchCamera() {
            useFrontCamera = !useFrontCamera;
            initCamera();
            showToast("Camera Switched", useFrontCamera ? "Front bunny cam" : "Rear cam active");
        }

        function capturePhoto() {
            const video = document.getElementById('camera-stream');
            const canvas = document.getElementById('snapshot-canvas');
            const previewImg = document.getElementById('preview-image');
            const modal = document.getElementById('caption-modal');

            if (video.readyState === video.HAVE_ENOUGH_DATA && !video.classList.contains('hidden')) {
                canvas.width = video.videoWidth || 720;
                canvas.height = video.videoHeight || 1280;
                const ctx = canvas.getContext('2d');
                if (useFrontCamera) {
                    ctx.translate(canvas.width, 0);
                    ctx.scale(-1, 1);
                }
                ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
                capturedDataUrl = canvas.toDataURL('image/png');
            } else {
                capturedDataUrl = 'https://images.unsplash.com/photo-1518709268805-4e9042af9f23?q=80&w=600&auto=format&fit=crop';
            }

            previewImg.src = capturedDataUrl;
            document.getElementById('modal-sticker-tag').innerText = activeSticker;
            modal.classList.remove('hidden');
            setTimeout(() => modal.classList.remove('opacity-0'), 10);
            document.getElementById('caption-input').value = '';
        }

        function closeCaptionModal() {
            const modal = document.getElementById('caption-modal');
            modal.classList.add('opacity-0');

                'https://images.unsplash.com/photo-1511556532299-8f662fc26c06?q=80&w=600&auto=format&fit=crop'
            ];
            const captions = ['Check this out! ✨', 'Miss you bunnies 💖', 'Working hard today 🎵', 'Yummy snack time 🍪'];

            const incoming = {
                id: Date.now(),
                author: sender.name,
                avatar: sender.avatar,
                photo: photos[Math.floor(Math.random() * photos.length)],
                caption: captions[Math.floor(Math.random() * captions.length)],
                timestamp: 'Just now',
                sticker: '✨ Super Shy'
            };

            moments.unshift(incoming);
            renderFeed();
            renderAlbum();
            showToast(`New Phoning Drop!`, `${sender.name} sent a photo widget.`);
        }

        function renderFeed() {
            const container = document.getElementById('phoning-chat-stream');
            container.innerHTML = moments.map(m => `
                <div class="bg-white border-2 border-purple-100 rounded-3xl p-4 shadow-[0_4px_20px_rgba(216,180,254,0.15)]">
                    <div class="flex items-center justify-between mb-3">
                        <div class="flex items-center space-x-2.5">
                            <div class="w-9 h-9 rounded-2xl bg-purple-100 flex items-center justify-center text-lg shadow-sm">
                                ${m.avatar}
                            </div>
                            <div>
                                <h4 class="font-fredoka font-bold text-xs text-purple-900">${m.author}</h4>
                                <span class="text-[10px] text-neutral-400">${m.timestamp}</span>
                            </div>
                        </div>
                        <span class="bg-purple-50 text-purple-600 font-fredoka text-[10px] px-2.5 py-1 rounded-full border border-purple-100">${m.sticker}</span>
                    </div>

                    <!-- Photo Message Card -->
                    <div class="relative w-full aspect-[4/5] rounded-2xl overflow-hidden bg-purple-50 border border-purple-100 shadow-inner">
                        <img src="${m.photo}" alt="Moment" class="w-full h-full object-cover">
                        <div class="absolute bottom-0 inset-x-0 p-3 bg-gradient-to-t from-black/70 via-black/20 to-transparent">
                            <p class="text-xs font-fredoka font-medium text-white text-center drop-shadow">${m.caption}</p>
                        </div>
                    </div>

                    <!-- Reactions -->
                    <div class="flex items-center justify-between mt-3 px-1">
                        <div class="flex space-x-2">
                            <button onclick="reactMoment(${m.id}, '💖')" class="bg-purple-50 hover:bg-purple-100 border border-purple-100 px-3 py-1 rounded-full text-xs text-purple-700 font-fredoka active:scale-95 transition">💖 Love</button>
                            <button onclick="reactMoment(${m.id}, '🐰')" class="bg-purple-50 hover:bg-purple-100 border border-purple-100 px-3 py-1 rounded-full text-xs text-purple-700 font-fredoka active:scale-95 transition">🐰 Bunny</button>
                        </div>
                        <span class="text-[10px] text-neutral-400 font-medium">Widget Active</span>
                    </div>
                </div>
            `).join('');
        }

        function renderAlbum() {
            const container = document.getElementById('album-grid');
            document.getElementById('album-count').innerText = moments.length;
            container.innerHTML = moments.map(m => `
                <div class="relative aspect-square rounded-2xl overflow-hidden border-2 border-purple-100 shadow-sm bg-white group">
                    <img src="${m.photo}" alt="Album item" class="w-full h-full object-cover group-hover:scale-105 transition duration-300">
                    <div class="absolute inset-x-0 bottom-0 p-2 bg-gradient-to-t from-black/80 to-transparent">
                        <p class="text-[10px] font-fredoka text-white truncate">${m.caption}</p>
                    </div>
                </div>
            `).join('');
        }

        function reactMoment(id, emoji) {
            showToast("Reaction Sent!", `You sent ${emoji} to the chat.`);
        }

        function showToast(title, desc) {
            const toast = document.getElementById('toast');
            document.getElementById('toast-title').innerText = title;
            document.getElementById('toast-desc').innerText = desc;
            toast.classList.remove('-translate-y-24', 'opacity-0');
            setTimeout(() => {
                toast.classList.add('-translate-y-24', 'opacity-0');
            }, 3000);
        }
